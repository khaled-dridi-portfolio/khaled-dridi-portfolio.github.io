> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Mohamed Khaled Dridi's static portfolio site with four audience presets (`/`, `/freelance/`, `/job/`, `/gamedev/`), Markdown-driven projects (draftable), two visual themes, and automatic deployment to GitHub Pages.

**Architecture:** Astro 7 static site. Projects and experience are Astro content collections (glob loader, Zod 4 schemas). Presets are plain TypeScript config consumed by one `PresetPage` component; four thin page files pick a preset. Themes are CSS custom properties switched by `data-theme` / `data-scheme` on `<html>`; project and catalog pages pick their theme in the browser from `?from=<preset>` because static builds have no query string at build time. Pure logic (ordering, dates, video parsing, schema rules, presets) lives in `src/lib` / `src/config` and is unit-tested with Playwright's test runner; pages are tested end-to-end with Playwright against `astro preview` of a fixture-enabled build in `dist-test/`.

**Tech Stack:** Astro 7.3.5, TypeScript 6.0.3, @astrojs/check 0.9.10, @playwright/test 1.63.0, Fontsource font packages 5.3.0, GitHub Actions (`withastro/action@v6`, `actions/deploy-pages@v5`), Node ≥ 22.12 (local: 24.14).

```ts
export const REQUIRED_WHEN_PUBLISHED = ['summary', 'type', 'role', 'startDate'] as const;

/** Fields a project must have once `draft` is false. Drafts may be incomplete. */
export function missingPublishedFields(data: Record<string, unknown>): string[] {
  if (data.draft === true) return [];
  const missing: string[] = REQUIRED_WHEN_PUBLISHED.filter((k) => data[k] === undefined || data[k] === '');
  if (!Array.isArray(data.tags) || data.tags.length === 0) missing.push('tags');
  return missing;
}
```

```ts
import type { Category } from './types';

export interface ProjectLike {
  id: string;
  data: { category: Category; startDate?: Date; featured: boolean; draft: boolean; tags: string[] };
}

export function byNewest(a: ProjectLike, b: ProjectLike): number {
  return (b.data.startDate?.getTime() ?? 0) - (a.data.startDate?.getTime() ?? 0);
}

export function publishedProjects<T extends ProjectLike>(projects: T[]): T[] {
  return projects.filter((p) => !p.data.draft).sort(byNewest);
}

export function selectForPreset<T extends ProjectLike>(
  projects: T[],
  opts: { categories: readonly Category[]; order: readonly string[] },
): T[] {
  const visible = publishedProjects(projects).filter((p) => opts.categories.includes(p.data.category));
  const pinned = opts.order.flatMap((id) => visible.filter((p) => p.id === id));
  const rest = visible.filter((p) => !opts.order.includes(p.id));
  return [...pinned, ...rest.filter((p) => p.data.featured), ...rest.filter((p) => !p.data.featured)];
}

export function allTags(projects: ProjectLike[]): string[] {
  return [...new Set(projects.flatMap((p) => p.data.tags))].sort((a, b) => a.localeCompare(b));
}

const monthYear = new Intl.DateTimeFormat('en', { month: 'short', year: 'numeric', timeZone: 'UTC' });

export function formatDateRange(start: Date | undefined, end: Date | undefined, presentLabel: string): string {
  if (!start) return '';
  const s = monthYear.format(start);
  if (!end) return `${s} – ${presentLabel}`;
  const e = monthYear.format(end);
  return s === e ? s : `${s} – ${e}`;
}
```

```ts
export type VideoSource = { kind: 'youtube'; id: string } | { kind: 'file'; src: string };

const YT_ID = /^[A-Za-z0-9_-]{11}$/;
const YT_HOSTS = new Set(['youtube.com', 'www.youtube.com', 'm.youtube.com', 'youtube-nocookie.com', 'www.youtube-nocookie.com']);

/** Accepts a YouTube URL or a local path starting with "/". Anything else returns null. */
export function parseVideo(value: string | undefined): VideoSource | null {
  if (!value) return null;
  if (value.startsWith('/')) return { kind: 'file', src: value };
  let url: URL;
  try {
    url = new URL(value);
  } catch {
    return null;
  }
  let id: string | null = null;
  if (url.hostname === 'youtu.be') {
    id = url.pathname.slice(1).split('/')[0] ?? null;
  } else if (YT_HOSTS.has(url.hostname)) {
    if (url.pathname === '/watch') id = url.searchParams.get('v');
    else id = url.pathname.match(/^\/(?:embed|shorts|live)\/([^/]+)/)?.[1] ?? null;
  }
  return id && YT_ID.test(id) ? { kind: 'youtube', id } : null;
}

export function youtubeEmbedUrl(id: string): string {
  return `https://www.youtube-nocookie.com/embed/${id}?autoplay=1&rel=0`;
}

export function youtubeWatchUrl(id: string): string {
  return `https://www.youtube.com/watch?v=${id}`;
}
```

```ts
        type: z.enum(PROJECT_TYPES).optional(),
        role: z.string().optional(),
```

```js
// Generates solid-colour PNGs used only by test fixtures. Run: node tests/fixtures/make-fixture-images.mjs
import { mkdirSync, writeFileSync } from 'node:fs';
import { deflateSync, crc32 } from 'node:zlib';

function chunk(type, data) {
  const len = Buffer.alloc(4);
  len.writeUInt32BE(data.length);
  const body = Buffer.concat([Buffer.from(type, 'ascii'), data]);
  const crc = Buffer.alloc(4);
  crc.writeUInt32BE(crc32(body) >>> 0);
  return Buffer.concat([len, body, crc]);
}

function png(width, height, [r, g, b]) {
  const row = Buffer.alloc(1 + width * 3);
  for (let x = 0; x < width; x++) row.set([r, g, b], 1 + x * 3);
  const raw = Buffer.concat(Array.from({ length: height }, () => row));
  const ihdr = Buffer.alloc(13);
  ihdr.writeUInt32BE(width, 0);
  ihdr.writeUInt32BE(height, 4);
  ihdr.set([8, 2, 0, 0, 0], 8);
  return Buffer.concat([
    Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]),
    chunk('IHDR', ihdr),
    chunk('IDAT', deflateSync(raw)),
    chunk('IEND', Buffer.alloc(0)),
  ]);
}
```

```ts
// Keep the storage key in sync with the inline script in BaseLayout.astro.
const SCHEME_KEY = 'scheme';

function storeScheme(value: string): void {
  try {
    localStorage.setItem(SCHEME_KEY, value);
  } catch {
    // Storage blocked (private mode, disabled cookies): the choice just won't persist.
  }
}

export function initThemeToggle(): void {
  const root = document.documentElement;
  document.querySelectorAll<HTMLButtonElement>('[data-scheme-toggle]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const next = root.dataset.scheme === 'light' ? 'dark' : 'light';
      root.dataset.scheme = next;
      storeScheme(next);
    });
  });
}

/** Points "home" links at the preset the visitor came from and carries ?from= onto project links. */
export function applyFromPreset(): void {
  const root = document.documentElement;
  const from = root.dataset.from;
  if (!from) return;
  const paths = JSON.parse(root.dataset.presetPaths ?? '{}') as Record<string, string>;
  if (!Object.hasOwn(paths, from)) return;
  const home = paths[from];
  document.querySelectorAll<HTMLAnchorElement>('a[data-preset-home]').forEach((a) => {
    a.setAttribute('href', home + (a.dataset.hash ?? ''));
  });
  document.querySelectorAll<HTMLAnchorElement>('a[data-keep-from]').forEach((a) => {
    const url = new URL(a.getAttribute('href') ?? '', location.href);
    url.searchParams.set('from', from);
    a.setAttribute('href', url.pathname + url.search + url.hash);
  });
}
```

```ts
export function initCategoryFilters(): void {
  document.querySelectorAll<HTMLElement>('[data-project-grid]').forEach((grid) => {
    const group = grid.querySelector<HTMLElement>('.filter-chips');
    if (!group) return;
    group.hidden = false;
    const chips = group.querySelectorAll<HTMLButtonElement>('[data-filter]');
    const cards = grid.querySelectorAll<HTMLElement>('[data-project-card]');
    chips.forEach((chip) => {
      chip.addEventListener('click', () => {
        const filter = chip.dataset.filter;
        chips.forEach((c) => c.setAttribute('aria-pressed', String(c === chip)));
        cards.forEach((card) => {
          card.hidden = filter !== 'all' && card.dataset.category !== filter;
        });
      });
    });
  });
}

export function initPreviewClips(): void {
  if (!matchMedia('(hover: hover)').matches || matchMedia('(prefers-reduced-motion: reduce)').matches) return;
  document.querySelectorAll<HTMLVideoElement>('video[data-preview-clip]').forEach((video) => {
    const card = video.closest<HTMLElement>('[data-project-card]');
    if (!card) return;
    card.addEventListener('mouseenter', () => {
      video.play().catch(() => {
        /* autoplay refused or file missing: keep the still image */
      });
    });
    card.addEventListener('mouseleave', () => {
      video.pause();
      video.currentTime = 0;
    });
  });
}
```

```ts
export function initCopyButtons(): void {
  document.querySelectorAll<HTMLButtonElement>('[data-copy]').forEach((btn) => {
    const original = btn.textContent;
    btn.addEventListener('click', async () => {
      const value = btn.dataset.copy ?? '';
      try {
        await navigator.clipboard.writeText(value);
      } catch {
        // Clipboard unavailable (insecure context or denied): open the mail client instead.
        window.location.href = `mailto:${value}`;
        return;
      }
      btn.textContent = btn.dataset.copiedLabel ?? original;
      setTimeout(() => {
        btn.textContent = original;
      }, 2000);
    });
  });
}
```

```ts
export function initFacades(): void {
  document.querySelectorAll<HTMLElement>('[data-facade]').forEach((el) => {
    el.querySelector('[data-facade-load]')?.addEventListener(
      'click',
      () => {
        const iframe = document.createElement('iframe');
        iframe.src = el.dataset.embed ?? '';
        iframe.title = el.dataset.title ?? '';
        iframe.allow = 'accelerometer; autoplay; clipboard-write; encrypted-media; fullscreen; gyroscope; picture-in-picture';
        iframe.allowFullscreen = true;
        el.replaceChildren(iframe);
        el.classList.add('is-loaded');
      },
      { once: true },
    );
  });
}
```

```ts
export function initGalleries(): void {
  document.querySelectorAll<HTMLElement>('[data-gallery]').forEach((gallery) => {
    const items = [...gallery.querySelectorAll<HTMLAnchorElement>('[data-gallery-item]')];
    const dialog = gallery.querySelector<HTMLDialogElement>('[data-lightbox]');
    if (!dialog || items.length === 0) return;
    const img = dialog.querySelector<HTMLImageElement>('[data-lightbox-img]')!;
    const count = dialog.querySelector<HTMLElement>('[data-lightbox-count]')!;
    const prev = dialog.querySelector<HTMLButtonElement>('[data-lightbox-prev]')!;
    const next = dialog.querySelector<HTMLButtonElement>('[data-lightbox-next]')!;
    prev.hidden = next.hidden = items.length < 2;
    let index = 0;

    const show = (i: number) => {
      index = (i + items.length) % items.length;
      img.src = items[index].href;
      img.alt = items[index].querySelector('img')?.alt ?? '';
      count.textContent = `${index + 1} / ${items.length}`;
    };

    items.forEach((a, i) =>
      a.addEventListener('click', (e) => {
        e.preventDefault();
        show(i);
        dialog.showModal();
      }),
    );
    prev.addEventListener('click', () => show(index - 1));
    next.addEventListener('click', () => show(index + 1));
    dialog.querySelector('[data-lightbox-close]')?.addEventListener('click', () => dialog.close());
    dialog.addEventListener('keydown', (e) => {
      if (e.key === 'ArrowLeft') show(index - 1);
      if (e.key === 'ArrowRight') show(index + 1);
    });

    // Swipe on touch screens; a swipe must not also count as a backdrop click.
    let startX: number | null = null;
    let swiped = false;
    dialog.addEventListener('pointerdown', (e) => {
      startX = e.clientX;
      swiped = false;
    });
    dialog.addEventListener('pointerup', (e) => {
      if (startX === null) return;
      const dx = e.clientX - startX;
      startX = null;
      if (Math.abs(dx) > 50) {
        swiped = true;
```

```ts
import { t } from '../i18n';

const startTime = (el: HTMLElement) => (el.dataset.start ? Date.parse(el.dataset.start) : 0);

export function initCatalog(): void {
  const root = document.querySelector<HTMLElement>('[data-catalog]');
  const list = root?.querySelector<HTMLElement>('[data-catalog-list]');
  const controls = root?.querySelector<HTMLElement>('[data-catalog-controls]');
  if (!root || !list || !controls) return;
  controls.hidden = false;

  const cards = [...list.querySelectorAll<HTMLElement>('[data-project-card]')];
  const chips = [...controls.querySelectorAll<HTMLButtonElement>('[data-filter]')];
  const techSelect = controls.querySelector<HTMLSelectElement>('[data-tech-filter]')!;
  const sortSelect = controls.querySelector<HTMLSelectElement>('[data-sort]')!;
  const count = controls.querySelector<HTMLElement>('[data-count]')!;
  const noMatch = root.querySelector<HTMLElement>('[data-no-match]')!;
  let category = 'all';

  const apply = () => {
    const tech = techSelect.value;
    const direction = sortSelect.value === 'oldest' ? 1 : -1;
    const sorted = [...cards].sort((a, b) => direction * (startTime(a) - startTime(b)));
    let visible = 0;
    for (const card of sorted) {
      const matchesCategory = category === 'all' || card.dataset.category === category;
      const matchesTech = !tech || (card.dataset.tags ?? '').split('|').includes(tech);
      card.hidden = !(matchesCategory && matchesTech);
      if (!card.hidden) visible++;
      list.append(card);
    }
    count.textContent = t.projects.count(visible);
    noMatch.hidden = visible > 0;
  };

  chips.forEach((chip) =>
    chip.addEventListener('click', () => {
      category = chip.dataset.filter ?? 'all';
      chips.forEach((c) => c.setAttribute('aria-pressed', String(c === chip)));
      apply();
    }),
  );
  techSelect.addEventListener('change', apply);
  sortSelect.addEventListener('change', apply);
}
```

- [ ] **Step 3: Create `scripts/generate-og.mjs`**

```js
// Screenshots the top of each preset page at 1200x630 into public/og/<key>.png.
// Run with `npm run og` (builds first, then serves dist/ from a tiny static server).
import { createServer } from 'node:http';
import { readFile, stat, mkdir } from 'node:fs/promises';
import { extname, join, normalize } from 'node:path';
import { chromium } from '@playwright/test';

const DIST = 'dist';
const OUT = 'public/og';
const PORT = 4410;
const PAGES = { default: '/', freelance: '/freelance/', job: '/job/', gamedev: '/gamedev/' };
const TYPES = {
  '.html': 'text/html; charset=utf-8', '.css': 'text/css', '.js': 'text/javascript', '.svg': 'image/svg+xml',
  '.png': 'image/png', '.webp': 'image/webp', '.jpg': 'image/jpeg', '.woff2': 'font/woff2', '.woff': 'font/woff', '.pdf': 'application/pdf',
};

const server = createServer(async (req, res) => {
  try {
    const urlPath = decodeURIComponent(new URL(req.url, 'http://x').pathname);
    let file = normalize(join(DIST, urlPath));
    if (!file.startsWith(normalize(DIST))) throw new Error('outside dist');
    if ((await stat(file)).isDirectory()) file = join(file, 'index.html');
    res.writeHead(200, { 'content-type': TYPES[extname(file)] ?? 'application/octet-stream' });
    res.end(await readFile(file));
  } catch {
    res.writeHead(404).end('not found');
  }
});

await new Promise((resolve) => server.listen(PORT, resolve));
await mkdir(OUT, { recursive: true });
const browser = await chromium.launch();
try {
  const page = await browser.newPage({ viewport: { width: 1200, height: 630 }, colorScheme: 'dark' });
  for (const [key, path] of Object.entries(PAGES)) {
    await page.goto(`http://localhost:${PORT}${path}`, { waitUntil: 'networkidle' });
    await page.evaluate(() => document.fonts.ready);
    await page.screenshot({ path: `${OUT}/${key}.png` });
    console.log(`wrote ${OUT}/${key}.png`);
  }
} finally {
  await browser.close();
  server.close();
}
```
